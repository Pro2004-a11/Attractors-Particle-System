# 1. Duffing Oscillator Equations
def duffing_deriv(state, t, gamma, delta, alpha, beta, omega):
    x, v = state
    # x'' + delta*x' + alpha*x + beta*x^3 = gamma*cos(omega*t)
    dxdt = v
    dvdt = gamma * np.cos(omega * t) - delta * v - alpha * x - beta * x**3
    return [dxdt, dvdt]

delta = 0.3    #  (Damping)
